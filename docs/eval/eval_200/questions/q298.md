```sql
SELECT b.brand_name, COUNT(DISTINCT e.supplier_name) AS supplier_count, COUNT(*) AS expense_lines FROM ins_mc_invexpenses e JOIN ins_mc_invcalc ic ON e.doc_id=ic.invcalc_id JOIN ins_opis o ON o.opis_id=ic.opis_id JOIN ins_cars c ON c.car_id=o.car_id JOIN srv_mc_horizon_brand b ON c.car_mc_brand=b.brand_id WHERE e.supplier_name IS NOT NULL AND e.supplier_name != '' GROUP BY b.brand_name ORDER BY supplier_count DESC
```

```sql
  T6.brand_name,
  count(DISTINCT T1.supplier_name) AS distinct_supplier_count
FROM ins_mc_invexpenses AS T1
INNER JOIN ins_mc_invitems AS T2
  ON T1.item_id = T2.item_id
INNER JOIN ins_mc_invcalc AS T3
  ON T2.invcalc_id = T3.invcalc_id
INNER JOIN ins_opis AS T4
  ON T3.opis_id = T4.opis_id
INNER JOIN ins_cars AS T5
  ON T4.car_id = T5.car_id
INNER JOIN srv_mc_horizon_brand AS T6
  ON T5.car_mc_brand = T6.brand_id
GROUP BY
  T6.brand_name
ORDER BY
  distinct_supplier_count DESC;
```
