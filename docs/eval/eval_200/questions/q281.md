```sql
WITH ClaimCosts AS (
  SELECT
    T3.total_cost
  FROM ins_damages AS T1
  INNER JOIN ins_opis AS T2
    ON T1.damage_id = T2.damage_id
  INNER JOIN ins_mc_invcalc AS T3
    ON T2.opis_id = T3.opis_id
  WHERE
    T3.is_active = 1 AND T3.total_cost > 0
),
P90_Threshold AS (
  SELECT
    total_cost AS p90_value
  FROM (
    SELECT
      total_cost,
      ROW_NUMBER() OVER (ORDER BY total_cost) AS rn,
      COUNT(*) OVER () AS total_rows
    FROM ClaimCosts
  ) AS sub
  WHERE
    rn = CEIL(0.9 * total_rows)
)
SELECT
  (SELECT p90_value FROM P90_Threshold) AS p90_threshold,
  COUNT(*) AS claims_above_p90
FROM ClaimCosts
WHERE
  total_cost > (SELECT p90_value FROM P90_Threshold);
```
