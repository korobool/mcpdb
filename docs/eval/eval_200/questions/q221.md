## Ground Truth

### SQL
```sql
WITH latest AS (SELECT ic.* FROM ins_mc_invcalc ic INNER JOIN (SELECT opis_id, type, MAX(invcalc_id) AS max_id FROM ins_mc_invcalc GROUP BY opis_id, type) lt ON lt.opis_id=ic.opis_id AND lt.type=ic.type AND lt.max_id=ic.invcalc_id), mc AS (SELECT opis_id, total_cost FROM latest WHERE type='mc'), n24 AS (SELECT opis_id, total_cost FROM latest WHERE type='n24') SELECT d.damage_typeid, AVG(mc.total_cost - n24.total_cost) AS avg_diff, COUNT(*) AS claims FROM mc JOIN n24 ON n24.opis_id=mc.opis_id JOIN ins_opis o ON o.opis_id=mc.opis_id JOIN ins_damages d ON d.damage_id=o.damage_id GROUP BY d.damage_typeid ORDER BY avg_diff DESC
```

```sql
SELECT
  d.damage_typeid,
  AVG(
    CASE
      WHEN ic.type = 'MC' THEN ic.total_cost
    END
  ) AS avg_mc_cost,
  AVG(
    CASE
      WHEN ic.type = 'N24' THEN ic.total_cost
    END
  ) AS avg_n24_cost
FROM ins_mc_invcalc AS ic
JOIN ins_opis AS o
  ON ic.opis_id = o.opis_id
JOIN ins_damages AS d
  ON o.damage_id = d.damage_id
WHERE
  ic.type IN ('MC', 'N24')
GROUP BY
  d.damage_typeid
ORDER BY
  d.damage_typeid;
```
