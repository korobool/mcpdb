### SQL
```sql
WITH latest_mc AS (SELECT ic.* FROM ins_mc_invcalc ic INNER JOIN (SELECT opis_id, MAX(invcalc_id) AS max_id FROM ins_mc_invcalc WHERE type='mc' GROUP BY opis_id) lm ON lm.opis_id=ic.opis_id AND lm.max_id=ic.invcalc_id) SELECT AVG(mc.total_cost) AS avg_mc_estimate, AVG(d.total_cost) AS avg_damage_total FROM latest_mc mc JOIN ins_opis o ON o.opis_id=mc.opis_id JOIN ins_damages d ON d.damage_id=o.damage_id
```
