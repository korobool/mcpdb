WITH costs_by_claim AS (
    SELECT
        opis_id,
        SUM(CASE WHEN type = 'MC' THEN total_cost ELSE 0 END) AS mc_cost,
        SUM(CASE WHEN type = 'N24' THEN total_cost ELSE 0 END) AS n24_cost
    FROM
        ins_mc_invcalc
    WHERE
        type IN ('MC', 'N24')
    GROUP BY
        opis_id
    HAVING
        SUM(CASE WHEN type = 'MC' THEN 1 ELSE 0 END) > 0
        AND SUM(CASE WHEN type = 'N24' THEN 1 ELSE 0 END) > 0

SELECT
    AVG(mc_cost - n24_cost) AS avg_cost_difference
FROM
    costs_by_claim;
