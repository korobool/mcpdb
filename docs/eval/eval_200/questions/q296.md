## Ground Truth

### SQL
```sql
WITH latest_mc AS (SELECT ic.* FROM ins_mc_invcalc ic INNER JOIN (SELECT opis_id, MAX(invcalc_id) AS max_id FROM ins_mc_invcalc WHERE type='mc' GROUP BY opis_id) lm ON lm.opis_id=ic.opis_id AND lm.max_id=ic.invcalc_id), dmg_lines AS (SELECT opis_id, COUNT(*) AS damage_lines FROM ins_opis_rows GROUP BY opis_id) SELECT b.brand_name, AVG(mc.total_cost / NULLIF(l.damage_lines, 0)) AS avg_cost_per_line, COUNT(*) AS claims FROM latest_mc mc JOIN ins_opis o ON o.opis_id=mc.opis_id JOIN ins_cars c ON c.car_id=o.car_id JOIN srv_mc_horizon_brand b ON c.car_mc_brand=b.brand_id JOIN dmg_lines l ON l.opis_id=mc.opis_id GROUP BY b.brand_name ORDER BY avg_cost_per_line DESC
```

### Step 1 (SQL) — OK
```sql
SELECT
  T4.brand_name,
  AVG(T1.total_cost) AS average_estimated_cost
FROM ins_mc_invcalc AS T1
INNER JOIN ins_opis AS T2
  ON T1.opis_id = T2.opis_id
INNER JOIN ins_cars AS T3
  ON T2.car_id = T3.car_id
INNER JOIN srv_mc_horizon_brand AS T4
  ON T3.car_mc_brand = T4.brand_id
GROUP BY
  T4.brand_name
ORDER BY
  average_estimated_cost DESC;
```
