## Agent Execution

### Step 1 (SQL) — OK
```sql
WITH ClaimExpenseCounts AS (
  SELECT
    T2.opis_id,
    COUNT(T1.expense_id) AS expense_count
  FROM ins_mc_invexpenses AS T1
  JOIN ins_mc_invcalc AS T2
    ON T1.doc_id = T2.invcalc_id
  GROUP BY
    T2.opis_id
)
SELECT
  CASE
    WHEN expense_count BETWEEN 1 AND 10
    THEN '1-10'
    WHEN expense_count BETWEEN 11 AND 20
    THEN '11-20'
    WHEN expense_count BETWEEN 21 AND 50
    THEN '21-50'
    WHEN expense_count > 50
    THEN '50+'
  END AS expense_range,
  COUNT(opis_id) AS number_of_claims
FROM ClaimExpenseCounts
WHERE expense_count > 0
GROUP BY
  expense_range
ORDER BY
  CASE
    WHEN expense_range = '1-10'
    THEN 1
    WHEN expense_range = '11-20'
    THEN 2
    WHEN expense_range = '21-50'
    THEN 3
    WHEN expense_range = '50+'
    THEN 4
  END;
```
