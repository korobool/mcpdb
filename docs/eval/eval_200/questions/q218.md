### SQL
```sql
WITH latest AS (SELECT ic.* FROM ins_mc_invcalc ic INNER JOIN (SELECT opis_id, type, MAX(invcalc_id) AS max_id FROM ins_mc_invcalc GROUP BY opis_id, type) lt ON lt.opis_id=ic.opis_id AND lt.type=ic.type AND lt.max_id=ic.invcalc_id), mc AS (SELECT opis_id, paints_cost FROM latest WHERE type='mc'), n24 AS (SELECT opis_id, paints_cost FROM latest WHERE type='n24') SELECT AVG(mc.paints_cost) AS avg_mc_paint, AVG(n24.paints_cost) AS avg_n24_paint FROM mc JOIN n24 ON n24.opis_id=mc.opis_id
```

```sql
    SELECT
      opis_id,
      MAX(CASE WHEN type = 'MC' THEN paints_cost ELSE NULL END) AS mc_paint_cost,
      MAX(CASE WHEN type = 'N24' THEN paints_cost ELSE NULL END) AS n24_paint_cost
    FROM ins_mc_invcalc
    WHERE opis_id IN (
        SELECT opis_id
        FROM ins_mc_invcalc
        WHERE type IN ('MC', 'N24')
        GROUP BY opis_id
        HAVING COUNT(DISTINCT type) = 2
    )
    GROUP BY opis_id
)
SELECT
    pc.opis_id,
    pc.mc_paint_cost,
    pc.n24_paint_cost,
    m.mc_make_name AS car_make
FROM PairedCosts pc
JOIN ins_opis o ON pc.opis_id = o.opis_id
JOIN ins_cars c ON o.car_id = c.car_id
JOIN srv_mc_catalog_make m ON c.car_make = m.mc_make;
```
