## Ground Truth

### SQL
```sql
WITH latest_mc AS (SELECT ic.* FROM ins_mc_invcalc ic INNER JOIN (SELECT opis_id, MAX(invcalc_id) AS max_id FROM ins_mc_invcalc WHERE type='mc' GROUP BY opis_id) lm ON lm.opis_id=ic.opis_id AND lm.max_id=ic.invcalc_id) SELECT (YEAR(d.damage_date) - c.car_year) AS vehicle_age, AVG(mc.total_cost) AS avg_cost, COUNT(*) AS claims FROM latest_mc mc JOIN ins_opis o ON o.opis_id=mc.opis_id JOIN ins_damages d ON d.damage_id=o.damage_id JOIN ins_cars c ON c.car_id=o.car_id WHERE d.damage_date IS NOT NULL AND c.car_year > 0 AND YEAR(d.damage_date) >= c.car_year GROUP BY vehicle_age HAVING claims >= 3 ORDER BY vehicle_age
```
