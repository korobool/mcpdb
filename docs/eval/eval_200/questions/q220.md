```sql
WITH avg_costs_by_brand_type AS (
    SELECT
        c.car_mc_brand,
        i.type AS calculation_type,
        AVG(i.total_cost) AS average_cost
    FROM
        ins_cars AS c
    JOIN
        ins_opis AS o ON c.car_id = o.car_id
    JOIN
        ins_mc_invcalc AS i ON o.opis_id = i.opis_id
    WHERE
        i.type IN ('mc', 'n24')
    GROUP BY
        c.car_mc_brand, i.type
)
SELECT
    car_mc_brand,
    MAX(CASE WHEN calculation_type = 'mc' THEN average_cost END) AS mc_average_cost,
    MAX(CASE WHEN calculation_type = 'n24' THEN average_cost END) AS n24_average_cost,
    (MAX(CASE WHEN calculation_type = 'mc' THEN average_cost END) - MAX(CASE WHEN calculation_type = 'n24' THEN average_cost END)) AS cost_difference
FROM
    avg_costs_by_brand_type
GROUP BY
    car_mc_brand
HAVING
    COUNT(DISTINCT calculation_type) = 2
ORDER BY
    car_mc_brand;
```
