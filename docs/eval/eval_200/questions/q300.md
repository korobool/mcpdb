## Ground Truth

### SQL
```sql
WITH latest_mc AS (SELECT ic.* FROM ins_mc_invcalc ic INNER JOIN (SELECT opis_id, MAX(invcalc_id) AS max_id FROM ins_mc_invcalc WHERE type='mc' GROUP BY opis_id) lm ON lm.opis_id=ic.opis_id AND lm.max_id=ic.invcalc_id), dmg_lines AS (SELECT opis_id, COUNT(*) AS damage_lines FROM ins_opis_rows GROUP BY opis_id), repairs AS (SELECT opis_id, AVG(DATEDIFF(end_repair, start_repair)) AS avg_repair_days FROM ins_invrepair WHERE start_repair IS NOT NULL AND end_repair IS NOT NULL GROUP BY opis_id) SELECT b.brand_name, COUNT(*) AS claims, AVG(mc.total_cost) AS avg_mc_cost, AVG(l.damage_lines) AS avg_lines, AVG(rep.avg_repair_days) AS avg_repair_days FROM latest_mc mc JOIN ins_opis o ON o.opis_id=mc.opis_id JOIN ins_cars c ON c.car_id=o.car_id JOIN srv_mc_horizon_brand b ON c.car_mc_brand=b.brand_id LEFT JOIN dmg_lines l ON l.opis_id=mc.opis_id LEFT JOIN repairs rep ON rep.opis_id=mc.opis_id GROUP BY b.brand_name HAVING claims >= 5 ORDER BY claims DESC
```

### Step 15 (SQL) — OK
```sql
WITH damage_lines_agg AS (
    -- Pre-aggregate the number of damage lines for each claim to ensure accurate averaging
    SELECT
        opis_id,
        COUNT(opis_rowid) AS damage_line_count
    FROM ins_opis_rows
    GROUP BY
        opis_id
),
active_costs AS (
    -- Select the cost from only the active calculation for each claim to avoid duplicates
    SELECT
        opis_id,
        total_cost
    FROM ins_mc_invcalc
    WHERE is_active = 1
)
SELECT
    T1.car_mc_brand AS brand_name,
    COUNT(DISTINCT T2.opis_id) AS total_claims,
    AVG(T4.total_cost) AS average_mc_cost,
    AVG(T3.damage_line_count) AS average_damage_lines,
    AVG(DATEDIFF(T5.end_repair, T5.start_repair)) AS average_repair_duration
FROM
    ins_cars AS T1
INNER JOIN
    ins_opis AS T2 ON T1.car_id = T2.car_id
LEFT JOIN
    damage_lines_agg AS T3 ON T2.opis_id = T3.opis_id
LEFT JOIN
    active_costs AS T4 ON T2.opis_id = T4.opis_id
LEFT JOIN
    ins_invrepair AS T5 ON T2.opis_id = T5.opis_id
WHERE
    T1.car_mc_brand IS NOT NULL AND T1.car_mc_brand != ''
GROUP BY
    T1.car_mc_brand
HAVING
    total_claims >= 5
ORDER BY
    total_claims DESC;
```
