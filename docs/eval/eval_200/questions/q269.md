## Ground Truth

### SQL
```sql
WITH latest_mc AS (SELECT ic.* FROM ins_mc_invcalc ic INNER JOIN (SELECT opis_id, MAX(invcalc_id) AS max_id FROM ins_mc_invcalc WHERE type='mc' GROUP BY opis_id) lm ON lm.opis_id=ic.opis_id AND lm.max_id=ic.invcalc_id) SELECT d.damage_place_country, AVG(mc.total_cost) AS avg_cost, COUNT(*) AS claims FROM latest_mc mc JOIN ins_opis o ON o.opis_id=mc.opis_id JOIN ins_damages d ON d.damage_id=o.damage_id WHERE d.damage_place_country IS NOT NULL AND d.damage_place_country != '' GROUP BY d.damage_place_country ORDER BY avg_cost DESC
```

### Step 1 (SQL) — OK
```sql
SELECT
  t3.damage_place_country,
  AVG(t1.total_cost) AS average_estimated_cost
FROM ins_mc_invcalc AS t1
INNER JOIN ins_opis AS t2
  ON t1.opis_id = t2.opis_id
INNER JOIN ins_damages AS t3
  ON t2.damage_id = t3.damage_id
WHERE
  t3.damage_place_country IS NOT NULL
  AND t3.damage_place_country != ''
GROUP BY
  t3.damage_place_country
ORDER BY
  average_estimated_cost DESC;
```
