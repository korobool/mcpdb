### Step 1 (SQL) — OK

```sql
SELECT
  CASE
    WHEN DATEDIFF(end_repair, start_repair) BETWEEN 0 AND 7 THEN '0-7 days'
    WHEN DATEDIFF(end_repair, start_repair) BETWEEN 8 AND 14 THEN '8-14 days'
    WHEN DATEDIFF(end_repair, start_repair) BETWEEN 15 AND 30 THEN '15-30 days'
    WHEN DATEDIFF(end_repair, start_repair) BETWEEN 31 AND 60 THEN '31-60 days'
    WHEN DATEDIFF(end_repair, start_repair) > 60 THEN '60+ days'
  END AS duration_bucket,
  COUNT(*) AS number_of_repairs
FROM ins_invrepair
WHERE start_repair IS NOT NULL
  AND end_repair IS NOT NULL
  AND DATEDIFF(end_repair, start_repair) >= 0
GROUP BY duration_bucket
ORDER BY
  MIN(DATEDIFF(end_repair, start_repair));
```
