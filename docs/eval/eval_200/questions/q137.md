WITH latest_mc AS (SELECT ic.* FROM ins_mc_invcalc ic INNER JOIN (SELECT opis_id, MAX(invcalc_id) AS max_id FROM ins_mc_invcalc WHERE type='mc' GROUP BY opis_id) lm ON lm.opis_id=ic.opis_id AND lm.max_id=ic.invcalc_id), dmg_lines AS (SELECT opis_id, COUNT(*) AS damage_lines FROM ins_opis_rows GROUP BY opis_id) SELECT l.damage_lines, AVG(mc.total_cost) AS avg_cost, COUNT(*) AS claims FROM latest_mc mc JOIN dmg_lines l ON l.opis_id=mc.opis_id GROUP BY l.damage_lines ORDER BY l.damage_lines

## Agent Execution

### Step 1 (SQL) — OK
```sql
WITH DamageLineCounts AS (
  SELECT
    opis_id,
    COUNT(opis_rowid) AS number_of_damage_lines
  FROM ins_opis_rows
  GROUP BY
    opis_id
)
SELECT
  T2.number_of_damage_lines,
  AVG(T1.total_cost) AS average_total_cost
FROM ins_mc_invcalc AS T1
JOIN DamageLineCounts AS T2
  ON T1.opis_id = T2.opis_id
WHERE
  T1.is_active = 1
GROUP BY
  T2.number_of_damage_lines
ORDER BY
  T2.number_of_damage_lines;
```
