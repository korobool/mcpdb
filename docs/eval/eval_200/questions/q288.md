## Ground Truth

### SQL
```sql
WITH latest_mc AS (SELECT ic.* FROM ins_mc_invcalc ic INNER JOIN (SELECT opis_id, MAX(invcalc_id) AS max_id FROM ins_mc_invcalc WHERE type='mc' GROUP BY opis_id) lm ON lm.opis_id=ic.opis_id AND lm.max_id=ic.invcalc_id) SELECT b.brand_name, ROUND(100*AVG(mc.newparts_cost/NULLIF(mc.total_cost,0)),1) AS parts_share_pct, ROUND(100*AVG(mc.paints_cost/NULLIF(mc.total_cost,0)),1) AS paint_share_pct, COUNT(*) AS claims FROM latest_mc mc JOIN ins_opis o ON o.opis_id=mc.opis_id JOIN ins_cars c ON c.car_id=o.car_id JOIN srv_mc_horizon_brand b ON c.car_mc_brand=b.brand_id WHERE mc.total_cost > 0 GROUP BY b.brand_name ORDER BY parts_share_pct DESC
```
