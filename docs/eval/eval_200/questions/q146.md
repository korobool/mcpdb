### Step 1 (SQL) — OK
```sql
WITH damaged_parts_counts AS (
    SELECT
        part_id,
        COUNT(*) AS damage_frequency
    FROM
        ins_opis_rows
    GROUP BY
        part_id
)
SELECT
    COALESCE(pf.part_name, pi.part_name, po.part_name) AS part_name,
    dpc.damage_frequency
FROM
    damaged_parts_counts dpc
LEFT JOIN
    ins_partsfree pf ON dpc.part_id = pf.part_id
LEFT JOIN
    ins_partsindex pi ON dpc.part_id = pi.part_id
LEFT JOIN
    ins_partsomc po ON dpc.part_id = po.part_id
WHERE
    COALESCE(pf.part_name, pi.part_name, po.part_name) IS NOT NULL
ORDER BY
    dpc.damage_frequency DESC
LIMIT 10;
```
