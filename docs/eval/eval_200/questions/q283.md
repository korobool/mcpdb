    ins_cars AS T2 ON T1.car_id = T2.car_id

    ins_damages AS T3 ON T1.damage_id = T3.damage_id

    ins_mc_invcalc AS T4 ON T1.opis_id = T4.opis_id
LEFT JOIN
    ins_opis_rows AS T5 ON T1.opis_id = T5.opis_id
WHERE
    T4.is_active = 1
GROUP BY
    T1.opis_id, T2.car_mc_brand, T3.damage_typeid, T4.total_cost
